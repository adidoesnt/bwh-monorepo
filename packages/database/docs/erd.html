    user {
        text id PK
        text email UK
        text role
        text status
        text timezone
    }
    coach_profile {
        text id PK
        text user_id FK
        text slug UK
        text timezone
    }
    availability_slot {
        text id PK
        text coach_id FK
        int weekday
        int start_min
        int end_min
    }
    booking {
        text id PK
        text client_id FK
        text coach_id FK
        text package_purchase_id FK
        text type
        text status
        timestamptz starts_at
    }
    package {
        text id PK
        text coach_id FK
        int session_count
        int session_length_min
        int price_per_session_cents
        int validity_days
    }
    package_purchase {
        text id PK
        text client_id FK
        text package_id FK
        int sessions_granted
        timestamptz expires_at
    }
    session_ledger_entry {
        text id PK
        text client_id FK
        text purchase_id FK
        text booking_id FK
        int delta
        text reason
    }
    invoice {
        text id PK
        text number UK
        text client_id FK
        text package_id FK
        text purchase_id FK
        text booking_id FK
        text status
        text stripe_checkout_session_id UK
        text stripe_payment_intent_id
    }
    intake_response {
        text id PK
        text client_id FK "unique"
        boolean parq_flag
    }
    progress_entry {
        text id PK
        text client_id FK
        text metric
        date recorded_on
    }
    measurement {
        text id PK
        text client_id FK
        text kind
        date taken_on
    }
    chat_message {
        text id PK
        text client_id FK
        text sender
    }
